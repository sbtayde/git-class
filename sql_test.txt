
--------------------------------------------------
PART 1: CREATE DATABASE AND TABLE
--------------------------------------------------

CREATE DATABASE college;

USE college;

CREATE TABLE students (
    student_id INT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    city VARCHAR(30),
    age INT,
    course VARCHAR(30),
    marks INT,
    fees INT,
    gender VARCHAR(10)
);
